method IsNonPrime (n : Nat) return (result : Bool)
  ensures result = true $\leftrightarrow$ $\neg$isPrime n
  do
    if n $\leq$ 1 then return true
    let mut i : Nat := 2
    let mut ret : Bool := false
    while i * i $\leq$ n
    invariant $\neg$ret $\leftrightarrow$ $\forall$d, 2 $\leq$ d $\land$ d < i $\to$ n % d $\neq$ 0
    invariant i $\geq$ 2 $\land$ (i - 1) * (i - 1) $\leq$ n
    do
      if n % i = 0 then ret := true
      i := i + 1
    return ret

prove_correct IsNonPrime by
  loom_solve  -- automated VC discharge
  ...         -- remaining interactive proof
\end{lstlisting}

\begin{frame}{Specification generation}
\stageBadge{1}
\setlength{\parskip}{0pt}
{\large\textbf{Step 2: Test the specification against the examples}\par}
\vspace{2mm}
Proving every example is expensive, so we \textbf{\color{mLightBrown}search for counterexamples} with property-based testing (PBT).
\par\vspace{2mm}
\centering
{\footnotesize\paperCodeFont\begin{tabular}{@{}l@{}}
\kw{def} precondition (arr : Array Int) := arr.size > 0\\
\textcolor{codeComment}{-- postcondition}\\
\kw{def} isSorted (arr : Array Int) (res : Array Int) :=\\
\hspace*{1em}$\forall$ i j, i < j $\to$ j < res.size $\to$ res[i]! $\leq$ res[j]!
\end{tabular}\par}
\vspace{2mm}
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{@{}l@{\hspace{5mm}}l@{\hspace{7mm}}c@{\hspace{7mm}}c@{}}
& & \small\texttt{[1,3,2]} $\mapsto$ \texttt{[1,2,3]} & \small\texttt{[]} $\mapsto$ \texttt{[]}\\
\hline
\uncover<2->{\textbf{\color{teal}Input validity}} &
  \uncover<2->{\small Does the input satisfy \texttt{precondition}?} &
  \uncover<2->{\yes} & \uncover<2->{\no}\\
\uncover<3->{\textbf{\color{teal}Expected output}} &
  \uncover<3->{\small Does the output satisfy the postcondition?} &
  \uncover<3->{\yes} & \uncover<3->{\yes}\\
\uncover<4->{\textbf{\color{teal}Output uniqueness}} &
  \uncover<4->{\small Is every other output rejected?} &
  \uncover<4->{\textbf{?}} & \uncover<4->{--}
\end{tabular}

\vspace{2mm}
\uncover<2->{{\small\color{red!70!black}Sorting \texttt{[]} is a valid task, but \texttt{precondition} rejects it: it is too strong.\par}}
\end{frame}

\begin{frame}{Specification generation}
\stageBadge{1}

\vspace{2mm}
If the spec admits a wrong output, it is usually close to the expected one.\\
So PBT mutates the expected output and checks the postcondition again.

\begin{frame}{Invariant inference}
\stageBadge{3}
\setlength{\parskip}{0pt}
\vspace{4mm}
\pipelineFig{3}
\par\vspace{10mm}
\centering
\uncover<2->{The program is generated. To prove it correct, every loop needs \textbf{invariants},\\
facts that hold before and after every iteration.\par}
\vspace{3mm}
\uncover<3->{Finding inductive invariants is a \textbf{\color{mLightBrown}notoriously difficult} problem.\\[1mm]
How do we do this effectively?\par}
\end{frame}

\begin{frame}[fragile]{Invariant inference}
\stageBadge{3}
\setlength{\parskip}{0pt}
{\large\textbf{Step 1: The LLM proposes loop invariants}\par}
\vspace{4mm}
\begin{lrbox}{\programInvariantListing}
\begin{minipage}{0.86\textwidth}
\begin{lstlisting}[
  style=paperLean,
  basicstyle=\paperCodeFont\mdseries\color{black}\fontsize{9}{10.4}\selectfont,
  linebackgroundcolor={\color{white}\ifnum\value{lstnumber}>3 \ifnum\value{lstnumber}<7 \color{yellow!16}\fi\fi},
  linebackgroundheight=7.6pt,linebackgrounddepth=2.6pt
]
let mut res := arr
let mut n : Nat := 1
while n $\neq$ res.size
  invariant n_le: n $\leq$ res.size
  invariant sorted: isSorted arr res
  invariant same_elems: isPerm res arr
do
  -- insert res[n] into the sorted prefix (inner loop)
  ...
  n := n + 1
return res
\end{lstlisting}
\end{minipage}
\end{lrbox}
